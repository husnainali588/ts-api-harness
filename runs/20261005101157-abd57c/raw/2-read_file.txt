 101| /** Fallback error handler: problems keep their status, malformed JSON is 400, anything unknown is 500 without a stack. */
 102| export function errorHandler(err: unknown, req: Request, res: Response, next: NextFunction): void {
 103|   if (res.headersSent) {
 104|     next(err);
 105|     return;
 106|   }
 107|   if (err instanceof HttpProblem) {
 108|     sendProblem(res, problem(req, err.status, err.slug, err.title, err.detail));
 109|     return;
 110|   }
 111|   if (errorType(err) === 'entity.parse.failed') {
 112|     sendProblem(res, problem(req, 400, 'malformed-json', 'Malformed JSON body', 'The request body is not valid JSON.'));
 113|     return;
 114|   }
 115|   const status = clientStatus(err);
 116|   if (status !== undefined) {
 117|     sendProblem(res, problem(req, status, 'bad-request', 'Bad request', 'The request could not be processed.'));
 118|     return;
 119|   }
 120|   sendProblem(res, problem(req, 500, 'internal', 'Internal server error', 'An unexpected error occurred.'));
 121| }
