  75| function handlerFor(route: AnyRoute): RequestHandler {
  76|   return async (req: Request, res: Response) => {
  77|     const params = route.params === undefined ? undefined : parseInput(route.params, req.params, 'params');
  78|     const query = route.query === undefined ? undefined : parseInput(route.query, req.query, 'query');
  79|     const body = route.body === undefined ? undefined : parseInput(route.body, req.body, 'body');
  80|     const reply = await route.handler({ params, query, body });
  81|     const schema = route.responses[reply.status];
  82|     if (schema === undefined) throw new ResponseContractError(`${route.method} ${route.path}: no response schema for ${reply.status}`);
  83|     const checked = schema.safeParse(reply.body);
  84|     if (!checked.success) throw new ResponseContractError(`${route.method} ${route.path}: reply does not match responses[${reply.status}]`);
  85|     if (reply.location !== undefined) res.location(reply.location);
  86|     if (checked.data === undefined) {
  87|       res.status(reply.status).end();
  88|       return;
  89|     }
  90|     res.status(reply.status).json(checked.data);
  91|   };
  92| }
